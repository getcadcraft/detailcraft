ການເຊື່ອມຕໍ່ແຜ່ນປາຍ ເຊື່ອມແຜ່ນເຂົ້າກັບປາຍຄານ ແລະ ຍຶດມັນດ້ວຍໂບລເຂົ້າກັບປີກ ຫຼື ເອວຂອງເສົາ. DetailCraft ວາງແຜ່ນ, ຕັດຄານກັບຄືນ, ເຈາະຮູທີ່ກົງກັນໃນແຜ່ນ ແລະ ປີກເສົາ, ເພີ່ມໂບລ ແລະ ການເຊື່ອມ, ແລະ ກວດໄລຍະຂອບ ແລະ ໄລຍະຫ່າງ.

:::warning
DetailCraft ບໍ່ອອກແບບການເຊື່ອມຕໍ່. ການຕັ້ງຄ່າເປັນຄ່າເລີ່ມຕົ້ນໃຫ້ທ່ານປ່ຽນ; ບໍ່ມີຫຍັງກວດຄວາມແຂງແຮງ. ຄຳເຕືອນເລື່ອງຂອບ ແລະ ໄລຍະຫ່າງໃຊ້ຄ່າຕ່ຳສຸດຂອງ EN 1993-1-8 Table 3.3, ຊຶ່ງອາດຈະແຕກຕ່າງຈາກກົດທີ່ທ່ານໃຊ້.
:::

## ມັນເຊື່ອມຕໍ່ຫຍັງໄດ້

- ຄານ **ລະດັບ** (ປາຍທັງສອງຢູ່ລະດັບດຽວກັນ), ຕັ້ງຊື່ (ການໝຸນ 0), ມີໜ້າຕັດຮູບ I.
- ເຂົ້າໄປທີ່ **ປີກ** ຫຼື **ເອວ** ຂອງເສົາຮູບ I **ຕັ້ງ**, ຕັ້ງສາກກັບມັນ.
- ປາຍຄານຕ້ອງຢູ່ເທິງເສັ້ນຂອງເສົາ, ລະຫວ່າງດ້ານລຸ່ມ ແລະ ດ້ານເທິງຂອງເສົາ.

ຄານທີ່ອຽງ ຫຼື ລາດ ແລະ ໜ້າຕັດທີ່ບໍ່ແມ່ນຮູບ I ຖືກປະຕິເສດພ້ອມເຫດຜົນ: ໃຊ້ [ແຜ່ນໜ້າຕັດ](topic:shear-tabs) ສຳລັບສິ່ງເຫຼົ່ານັ້ນ. ຢູ່ເອວ, ແຜ່ນ ແລະ ປີກຄານຕ້ອງພໍດີລະຫວ່າງປີກເສົາ; DetailCraft ເຕືອນເມື່ອມັນບໍ່ພໍດີ.

:::tip
ທີ່ການໝຸນ 0, ປີກຂອງເສົາຫັນໄປຕາມທິດເສັ້ນຕາຂ່າຍທີ່ໃສ່ເລກ (x). ຄານທີ່ແລ່ນຈາກ A ໄປ B ພົບເອວ, ແລະ ໄດ້ແຜ່ນປາຍຢູ່ເອວ.
:::

## ເພີ່ມແຜ່ນປາຍໃສ່ຄານ

1. ເລືອກ {{ui:ເລືອກ}} ({{key:V}}) ແລ້ວຄລິກຄານ.
2. ເລືອກ {{ui:ແຜ່ນປາຍ}} ໃນກຸ່ມ {{ui:ເຊື່ອມຕໍ່}}, ຫຼື {{ui:ເພີ່ມແຜ່ນປາຍ}} ໃນແຜງ {{ui:ຄຸນສົມບັດ}}.
3. ທຸກປາຍຄານທີ່ວ່າງ ທີ່ພົບປີກ ຫຼື ເອວຂອງເສົາເປັນສາກ ຈະໄດ້ແຜ່ນ. ແຖບສະຖານະບອກວ່າ "ເພີ່ມແຜ່ນປາຍ 2 ແຜ່ນແລ້ວ. ເລືອກໜຶ່ງເພື່ອປ່ຽນແຜ່ນ, ໂບລ ແລະ ການເຊື່ອມ; ຄ້າຍຄືກັນ ຈະສຳເນົາໄປຈຸດຕໍ່ທີ່ຄ້າຍກັນ."
4. ແຜ່ນປາຍອັນທຳອິດທີ່ເພີ່ມໃໝ່ຈະຖືກເລືອກ ແລະ ການຕັ້ງຄ່າຂອງມັນຢູ່ໃນແຜງ {{ui:ຄຸນສົມບັດ}}.

ຖ້າບໍ່ມີຫຍັງເກີດຂຶ້ນ, ໃຫ້ອ່ານແຖບສະຖານະ:

- "ຄານນີ້ບໍ່ໄດ້ຢູ່ເທິງເສັ້ນເສົາ. ແຜ່ນປາຍຖືກວາງບ່ອນທີ່ປາຍຄານພົບເສົາຕັ້ງ."
- "ຄານນີ້ພົບເສົາເປັນມຸມ ຫຼື ອຽງ: ແຜ່ນປາຍຖືກສ້າງສຳລັບຈຸດຕໍ່ສາກ ແລະ ລະນາບ. ໃຊ້ແຜ່ນໜ້າຕັດ."
- "ປາຍທັງສອງມີການເຊື່ອມຕໍ່ຢູ່ແລ້ວ."

## ປ່ຽນແຜ່ນ, ໂບລ ແລະ ການເຊື່ອມ

|---|---|---|

ຢູ່ລຸ່ມການຕັ້ງຄ່າ ທ່ານເຫັນສິ່ງທີ່ການເຊື່ອມຕໍ່ສ້າງ:

- {{ui:ໂບລ}}: ຕົວຢ່າງ "8 × 20 × 70 A325 (grip 36.2)".
- {{ui:ການເຊື່ອມ}}: ຂະໜາດ, ບ່ອນ, ແລະ ຄວາມຍາວຂອງມັນ.
- {{ui:ແຜ່ນ}}: ຂະໜາດ ແລະ ເຄື່ອງໝາຍຊິ້ນສ່ວນຂອງມັນ.

### ວິທີຄິດໄລ່ຕົວເລກ

- ຄານຖືກຕັດກັບຄືນເທົ່າກັບເຄິ່ງຄວາມເລິກຂອງເສົາ (ຢູ່ເອວ: ເຄິ່ງຄວາມໜາຂອງເອວ) ບວກຄວາມໜາແຜ່ນ, ດັ່ງນັ້ນແຜ່ນຈະຢູ່ເທິງໜ້າ.
- ຄວາມສູງແຜ່ນເທົ່າກັບຄວາມເລິກຂອງຄານບວກສອງສ່ວນຍືດ.
- ຄວາມຍາວໂບລ = grip (ຄວາມໜາແຜ່ນ + ຄວາມໜາປີກ ຫຼື ເອວຂອງເສົາ) + ສ່ວນເຜື່ອ, ປັດຂຶ້ນເປັນ 5 mm ຕໍ່ໄປ.
- ຄວາມຍາວການເຊື່ອມທີ່ສະແດງແມ່ນຮອບໜ້າຕັດຂອງຄານ.

### ຄຳເຕືອນ

ການເຊື່ອມຕໍ່ຍັງສ້າງຢູ່, ແຕ່ອັນເຫຼົ່ານີ້ປາກົດຢູ່ລຸ່ມມັນ ແລະ ໃນແຖບ {{ui:ບັນຫາ}}:

- ແຖວໂບລ ຫຼື ໂບລໃກ້ຂອບແຜ່ນກວ່າ 1.2 × ເສັ້ນຜ່າສູນກາງຮູ.
- ໄລຍະແຖວຕ່ຳກວ່າ 2.2 × ເສັ້ນຜ່າສູນກາງຮູ, ຫຼື gauge ຕ່ຳກວ່າ 2.4 × ເສັ້ນຜ່າສູນກາງຮູ.
- ຮູທີ່ລົງໄປໃນເອວເສົາ ຫຼື ຮອຍໂຄ້ງຂອງມັນ, ຫຼື ອອກນອກປີກເສົາ.
- ແຜ່ນກວ້າງກວ່າປີກເສົາ, ຫຼື (ຢູ່ເອວ) ກວ້າງກວ່າຊ່ອງລະຫວ່າງປີກ; ປີກຄານທີ່ບໍ່ພໍດີລະຫວ່າງມັນ.
- ແຖວໂບລໃກ້ປີກຄານເກີນໄປ, ຫຼື gauge ແໜ້ນເກີນໄປສຳລັບເອວຄານ ແລະ ການເຊື່ອມຂອງມັນ.

## ສຳເນົາໄປຫາທຸກຈຸດຕໍ່ທີ່ຄ້າຍກັນ

1. ເລືອກການເຊື່ອມຕໍ່ທີ່ທ່ານຕັ້ງໄວ້.
2. ເລືອກ {{ui:ນຳໃຊ້ກັບຄ້າຍຄືກັນ (n)}} ໃນແຜງ {{ui:ຄຸນສົມບັດ}}, ຫຼື {{ui:ຄ້າຍຄືກັນ}} ໃນເມນູ {{ui:ເພີ່ມເຕີມ}} ຂອງກຸ່ມ {{ui:ເຊື່ອມຕໍ່}}. ຕົວເລກໃນວົງເລັບແມ່ນຈຳນວນຈຸດຕໍ່ທີ່ຈະໄດ້ມັນ.
3. ທຸກຈຸດຕໍ່ທີ່ມີໜ້າຕັດຄານ ແລະ ເສົາດຽວກັນ, ຄານເຂົ້າໃນໜ້າປະເພດດຽວກັນ, ແລະ ຍັງບໍ່ມີການເຊື່ອມຕໍ່, ຈະໄດ້ສຳເນົາທີ່ມີການຕັ້ງຄ່າດຽວກັນ. ແຖບສະຖານະບອກຈຳນວນ.

ສຳເນົາເປັນອິດສະຫຼະ: ການປ່ຽນໜຶ່ງໃນພາຍຫຼັງ ບໍ່ປ່ຽນອັນອື່ນ. ຈຸດຕໍ່ທີ່ມີການເຊື່ອມຕໍ່ແລ້ວຈະຖືກປະໄວ້; ລຶບອັນເຫຼົ່ານັ້ນກ່ອນ ຖ້າທ່ານຕ້ອງການໃຫ້ມັນຖືກແທນທີ່.

## ການຍ້າຍ ແລະ ການລຶບ

ການເຊື່ອມຕໍ່ຖືກສ້າງໃໝ່ຈາກການຕັ້ງຄ່າຂອງມັນທຸກຄັ້ງທີ່ໂມເດວປ່ຽນ, ດັ່ງນັ້ນການຍ້າຍ ຫຼື ປັບຂະໜາດຄານ ຫຼື ເສົາ ຈະນຳແຜ່ນປາຍໄປນຳ. ຖ້າການປ່ຽນແປງເຮັດໃຫ້ຈຸດຕໍ່ບໍ່ພໍດີອີກ, ການເຊື່ອມຕໍ່ຢູ່ໃນໂມເດວ ແລະ ແຖບ {{ui:ບັນຫາ}} ບອກເຫດຜົນ, ເຊັ່ນ "ປາຍຄານບໍ່ຢູ່ເທິງເສັ້ນເສົາອີກຕໍ່ໄປ."

ເພື່ອລຶບການເຊື່ອມຕໍ່, ເລືອກມັນ ແລ້ວກົດ {{key:Delete}}. ການລຶບຄານ ຫຼື ເສົາຈະລຶບການເຊື່ອມຕໍ່ຂອງມັນນຳເຊັ່ນກັນ.
