## ភាពខុសគ្នាដែលធ្វើឲ្យមនុស្សជំពប់

### សញ្ញាផ្លាស់ប្តូរដោយខ្លួនឯង

គ្មានជំហានដាក់លេខទេ។ សញ្ញាត្រូវបានគណនាឡើងវិញបន្ទាប់ពីគ្រប់ការផ្លាស់ប្តូរ ដូច្នេះវាតែងតែត្រឹមត្រូវ ប៉ុន្តែវាក៏អាចដាក់លេខឡើងវិញដែរ៖ បន្ថែមសសរ ហើយ c3 ម្សិលមិញអាចក្លាយជា c4 ឥឡូវនេះ។ គ្មាន "ប្រវត្តិលេខ" ដើម្បីរក្សាសញ្ញាចាស់ទេ។ បញ្ចប់ម៉ូដែល បន្ទាប់មកចេញ BOM ឯកសារ NC និងគំនូររោងជាងជាមួយគ្នាពីឯកសារដែលបានរក្សាទុកមួយ។

### ប៊ូតុងកណ្តាលកណ្តុរធ្វើឲ្យពង្រីក មិនអូស

នៅក្នុងទិដ្ឋភាព 3D៖ អូសខាងឆ្វេងដើម្បីបង្វិល អូសខាងស្តាំដើម្បីអូសទិដ្ឋភាព ផ្ទៃកង់ធ្វើឲ្យពង្រីក ហើយអូសដោយប៊ូតុងកណ្តាលក៏ពង្រីកដែរ។ ការកាន់ {{key:Shift}} ខណៈអូសខាងឆ្វេងក៏អូសទិដ្ឋភាពដែរ។ ផ្ទៃកង់ពង្រីកឆ្ពោះទៅកណ្តាលទិដ្ឋភាព មិនមែនទៅកាន់ទស្សន៍ទ្រនិចទេ។ គ្មានការជ្រើសចំណុចបង្វិលទេ៖ ម៉ូដែលបង្វិលជុំវិញកណ្តាលទិដ្ឋភាព។ សូមមើល [ផ្លាស់ទីជុំវិញទិដ្ឋភាព 3D](topic:navigate-3d)។

### ជួរខាងមុខ សសរ​ទាំងនេះ​ប្រឈមទៅមួយទិសដោយលំនាំដើម

នៅការបង្វិល 0 ជួរនៃសសរត្រូវបានតម្រឹមតាមទិសក្រឡាចត្រង្គលេខ។ ធ្នឹមដែលរត់ចន្លោះបន្ទាត់អក្សរចូលទៅក្នុងស្បៃ ហើយបន្ទះចុងត្រូវបានបដិសេធនៅទីនោះ ("ធ្នឹមចូលក្នុងស្បៃសសរមិនទាន់បានសាងសង់")។ បង្វិលសសរទាំងនោះ 90° មុនពេលបន្ថែមការភ្ជាប់។

### ការមើលទៅមុខ លើតែបន្ទាត់អក្សរ

{{ui:ការមើលទៅមុខ}} មើលតាមបន្ទាត់ក្រឡាចត្រង្គអក្សរតែប៉ុណ្ណោះ។ គ្មានទិដ្ឋភាពតាមបន្ទាត់លេខ គ្មានទិដ្ឋភាពដែលបានរក្សាទុក ហើយគ្មានប្លង់ការងារទេ។ ប្រើ {{ui:3D}} ឬ {{ui:បន្ថែមតាមក្រឡាចត្រង្គ}} សម្រាប់ស៊ុមលើបន្ទាត់លេខ។

### ការជ្រើសរើស ចម្លង និងផ្លាស់ទី

ជ្រើសរើសផ្នែកជាច្រើនជាមួយ {{key:Shift}}+ចុច ឬ {{key:Mod+A}} បន្ទាប់មកចម្លង កាត់ បិទភ្ជាប់ (នៅចំនុចប្រសព្វក្រឡាចត្រង្គ នៅការផ្លាស់ទីដែលវាយចូល ឬនៅកន្លែងដដែល) និងចម្លងពួកវា រួមទាំងការភ្ជាប់ផង ([ចម្លង បិទភ្ជាប់ និងចម្លងទ្វេ](topic:copy-paste))។ គ្មានការជ្រើសរើសដោយផ្ទាំងទេ គ្មានតម្រងការជ្រើសរើស គ្មានការផ្លាស់ទី ការឆ្លុះ ឬអារេ ហើយលក្ខណៈសម្បត្តិត្រូវបានផ្លាស់ប្តូរសមាជិកម្តងមួយនៅក្នុងផ្ទាំង {{ui:លក្ខណៈសម្បត្តិ}}។

### ក្រឡាចត្រង្គមិនទាញសមាជិកទៅជាមួយ

សមាជិករក្សាកូអរដោនេរបស់ពួកវាពេលអ្នកផ្លាស់ប្តូរចន្លោះក្រឡាចត្រង្គ; គ្មានអ្វីផ្លាស់ទីជាមួយបន្ទាត់ក្រឡាចត្រង្គទេ។ ផ្លាស់ប្តូរក្រឡាចត្រង្គមុនដាក់ដែក ឬកែសម្រួលកូអរដោនេចុងរបស់សមាជិកនីមួយៗ បន្ទាប់ពីនោះ។

### ការភ្ជាប់មិនត្រូវបានរចនា

ការភ្ជាប់ទាំងពីរដាក់បន្ទះ រន្ធ ប៊ូឡុង និងការផ្សារ ពីលេខដែលអ្នកផ្តល់។ គ្មានការពិនិត្យការរចនាណាមួយឡើយ មានតែការព្រមានចម្ងាយគែម និងចន្លោះ (ពី EN 1993-1-8 Table 3.3) ប៉ុណ្ណោះ។

### រន្ធមិនត្រូវបានកាត់ក្នុងទិដ្ឋភាព 3D

រន្ធបង្ហាញជាចុចងងឹត។ ពួកវាជារន្ធពិតក្នុងឯកសារ NC គំនូររោងជាង និងឯកសារ IFC។ ការកាត់ (copes) ត្រូវបានកាត់ក្នុងទិដ្ឋភាព។

### ទម្ងន់គឺសរុប

ទម្ងន់ដែកមិនដករន្ធ ឬការកាត់ទេ ហើយទម្ងន់ប៊ូឡុងមិនត្រូវបានរួមបញ្ចូល។

### មីលីម៉ែត្រតែប៉ុណ្ណោះ

អ្វីៗទាំងអស់ត្រូវបានវាយ និងបង្ហាញជាមីលីម៉ែត្រ សូម្បីតែសម្រាប់រូបរាង AISC។ ឈ្មោះប្រវត្តិរូបរក្សាទម្រង់ដែលបានបោះពុម្ពរបស់វា (W12x50, HSS6x6x1/4)។

### គ្មានគំនូរត្រូវគ្រប់គ្រង

គ្មានបញ្ជីគំនូរ គ្មានកម្មវិធីកែសម្រួលគំនូរ គ្មានប្លង់ទូទៅ ឬគំនូរផ្គុំទេ។ គំនូររោងជាងត្រូវបានធ្វើនៅពេលអ្នកស្នើសុំ ហើយទាញយក; ដើម្បីផ្លាស់ប្តូរវា ផ្លាស់ប្តូរម៉ូដែល ហើយធ្វើវាឡើងវិញ។

### វារត់ក្នុងកម្មវិធីរុករក

ការរក្សាទុកទាញយកឯកសារមួយ។ ផ្លូវកាត់ក្តារចុចដែលកម្មវិធីរុករកទុកសម្រាប់ខ្លួនវាមិនអាចប្រើបានទេ។ ម៉ូដែលរបស់អ្នកមិនចាកចេញពីកុំព្យូទ័ររបស់អ្នកឡើយ៖ សូមមើល [ឯកសាររបស់អ្នកទៅណា](common:your-files)។
