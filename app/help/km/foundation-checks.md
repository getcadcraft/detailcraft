ផ្ទាំង {{ui:Foundation}} ពិនិត្យគ្រឹះនៅក្នុងម៉ូដែលធៀបនឹងបន្ទុកសសរដែលអ្នកផ្ដល់ឱ្យពួកវា។

## ពិពណ៌នាដី

វាយ {{ui:Allowable bearing pressure}} និង {{ui:Base friction coefficient}}។ ទាំងពីរចាប់ផ្ដើមនៅ 0 ដែលមានន័យថាមិនទាន់បានបញ្ចូល៖ ការពិនិត្យនឹងបង្ហាញសម្ពាធជាកំណត់សម្គាល់ ហើយមិននិយាយថាវាឆ្លងកាត់ទេ។ សម្រាប់ការអង្គុយ វាយម៉ូឌុលដី {{ui:Es}} (MPa) អត្រាពואសុន និងការអង្គុយដែលអ្នកអនុញ្ញាត (25 mm ដើម្បីចាប់ផ្ដើម)។ សម្រាប់ម៉ាត់បត់បែន វាយម៉ូឌុលស្រទាប់ដី {{ui:ks}}។ ការបំពេញលើគ្រឹះ ទម្ងន់ឯកតានៃដី កត្តាសុវត្ថិភាពសម្រាប់ការរអិល និងការផ្ដួល (1.5 ដើម្បីចាប់ផ្ដើម) និងកម្លាំងផលិតនៃដែកក៏មាននៅទីនោះដែរ។

## ផ្ដល់បន្ទុក

គ្រឹះនីមួយៗមានបញ្ជីបន្ទុកសសរ។ ប្រើ {{ui:Add a column load}} ហើយបំពេញជួរសេវាកម្ម (សម្រាប់ការទ្រ ការរអិល និងការផ្ដួល) និងជួរដែលបានគុណ (សម្រាប់ការពិនិត្យកម្លាំង)៖ បន្ទុកបញ្ឈរ P ចុះក្រោម កម្លាំងផ្ដេក Vx និង Vy នៅលើផ្ទៃខាងលើគ្រឹះ ម៉ូម៉ង់ Mx និង My ទីតាំងរបស់សសរលើគ្រឹះ និងទំហំរបស់វា។

ឬដំណើរការផ្ទាំង {{ui:Analysis}} ហើយចុច {{ui:Take column loads from the Analysis}}៖ ការទ្រគ្រប់ដែលឈរលើគ្រឹះនីមួយៗក្លាយជាបន្ទុកសសរ ពីសំណុំសេវាកម្ម និងសំណុំដែលបានគុណដែលអ្នកជ្រើស។ My វិជ្ជមានផ្ទុកផ្នែក +x នៃគ្រឹះច្រើនជាង; Mx វិជ្ជមានផ្ទុកផ្នែក −y ច្រើនជាង។

## អ្វីដែលត្រូវបានពិនិត្យ

ពីបន្ទុកសេវាកម្ម ជាមួយទម្ងន់ផ្ទាល់របស់គ្រឹះ និងការបំពេញដែលបានបូក៖

- **ការទ្រ**៖ សម្ពាធនៅក្រោមគ្រឹះរឹង។ នៅក្នុងមួយភាគបីកណ្ដាល មូលដ្ឋានទាំងមូលប៉ះគ្នា; នៅខាងក្រៅវា ផ្នែកខ្លះនៃមូលដ្ឋានលើកឡើង ហើយកំពូលខ្ពស់ជាង។ លទ្ធផលដែលនៅក្រៅមូលដ្ឋានគឺបរាជ័យ។
- **ការរអិល**៖ កកិតគុណនឹងបន្ទុកបញ្ឈរ ធៀបនឹងកម្លាំងផ្ដេក។
- **ការផ្ដួល**៖ ចម្ងាយរបស់លទ្ធផលពីគែម។
- **ការអង្គុយ**៖ ការអង្គុយបត់បែននៃសម្ពាធមធ្យមលើកន្លែងពាក់កណ្ដាលអនន្តនៃម៉ូឌុលដែលអ្នកបានផ្ដល់ នៅកណ្ដាល និងនៅជ្រុង (ដំណោះស្រាយរបស់ Timoshenko និង Goodier សម្រាប់ចតុកោណបត់បាន) ធៀបនឹងដែនកំណត់។ វាជាព្រំដែនខាងលើសម្រាប់គ្រឹះរឹង ហើយមិនរាប់បញ្ចូលស្រទាប់ដីដែលមានជម្រៅកំណត់ ការបន្ថយដោយដីដែលជីកចេញ និងការបង្រួម។ បើគ្មានម៉ូឌុល វាបង្ហាញសម្ពាធជាកំណត់សម្គាល់។

ពីបន្ទុកសសរដែលបានគុណតែមួយ (ACI 318-19)៖

- **ការពត់** នៅមុខសសរ ជាមួយដែកអប្បបរមា (0.0018 សម្រាប់ដែក 420 MPa) និងដែកដែលអ្នកបានគូរ។
- **កម្លាំងកាត់មួយទិស** នៅ d ពីមុខ។
- **កម្លាំងទម្លុះ** នៅ d/2 ជុំវិញសសរនីមួយៗ ជាមួយម៉ូម៉ង់របស់សសរ។ សម្រាប់សសរនៅជិតគែម ផ្នែកសំខាន់ត្រូវបានកាត់ដោយគែមនៃគ្រឹះ (បីជ្រុងនៅគែម ពីរនៅជ្រុង ACI 318-19 22.6.4.1 និង αs ដែលតូចជាងក្នុងតារាង 22.6.5.2) ហើយម៉ូម៉ង់ជុំវិញផ្ចិតផ្នែកនោះរាប់បញ្ចូលការផ្លាស់ទីនៃបន្ទុក។
- **ការអភិវឌ្ឍ** នៃដែកហួសមុខសសរ ត្រង់ ឬជាមួយការពត់របស់វា។
- **ជម្រៅ** លើដែកខាងក្រោម (150 mm)។

គ្រឹះរួមគ្នា ឬម៉ាត់ត្រូវបានចាត់ទុកជារឹង ហើយកម្លាំងកាត់ និងម៉ូម៉ង់មកពីស្ថិតិលើការកាត់ឆ្លងកាត់ទទឹងពេញ។

## ម៉ាត់ផ្ដល់ផ្ទាល់ខ្លួនលើស្ព្រីង

ធីក {{ui:Analyse this footing as a flexible mat on springs}} នៅពេលគ្រឹះផ្ទុកសសរច្រើន ហើយគ្រឹះរឹងមិនសូវត្រឹមត្រូវ។ គ្រឹះនោះក្លាយជាចានមួយនៃធាតុបួនថ្នាំង នៅលើគ្រែស្ព្រីង (ដែល {{ui:ks}} ដែលអ្នកបានវាយ ចែកចេញជាចំណុចថ្នាំង) ហើយបន្ទុកនៃសសរនីមួយៗត្រូវបានចែកទៅចំណុចថ្នាំងនៃទីតាំងរបស់វាផ្ទាល់។ តារាងទីពីរផ្ដល់សម្ពាធដីធំបំផុត និងការអង្គុយ (និងភាពខុសគ្នារវាងធំបំផុត និងតូចបំផុត) ម៉ូម៉ង់ការពត់ក្នុងទិសនីមួយៗ ជាមួយដែកដែលត្រូវការ ការតានៅផ្នែកខាងលើធៀបនឹងកម្លាំងបេតុងធម្មតា (គ្រឹះគ្មានដែកខាងលើ) កម្លាំងកាត់មួយទិសនៅ d ពីសសរ និងការទម្លុះលើសម្ពាធស្ព្រីងនៅខ្លួនវាផ្ទាល់។ ទំព័រចំណុចត្រូវបានកែលម្អរហូតដល់សសរមួយមានទទឹងបីធាតុ; ម៉ាត់ធំត្រូវការពេលពីរបីវិនាទី។ ទម្ងន់ផ្ទាល់របស់ម៉ាត់ និងដីលើវាលុបចោលគ្នា ហើយត្រូវបានទុកចោល។

ការពិនិត្យដែលបរាជ័យទាំងអស់ស្ថិតនៅក្នុងផ្ទាំង {{ui:Problems}} ជាមួយមាត្រារបស់វា។

## ក្រុមបង្គោល និងជញ្ជាំងទប់ដី

{{ui:Add a pile group}} ដាក់ក្រឡាចត្រង្គនៃបង្គោលនៅក្រោមមួករឹង៖ វាយបន្ទុក និងម៉ូម៉ង់ និងអ្វីដែលបង្គោលមួយអាចទ្រាំបាននៅការបង្ហាប់ និងការទាញ (លេខរបស់ការរចនាបង្គោល មិនមែនលេខនៃផ្ទាំងនេះទេ)។ ប្រតិកម្មរបស់បង្គោលនីមួយៗគឺ P/n បូកនឹងចំណែករបស់វានៃម៉ូម៉ង់ (វិធីសាស្ត្រលីនេអ៊ែរ) ហើយការបង្ហាប់ធំបំផុត និងការទាញណាមួយត្រូវបានពិនិត្យ។

{{ui:Add a retaining wall}} ពិនិត្យជញ្ជាំងកាន់ជ្រុងមួយ លើការបំពេញដីកម្រិត ក្នុងមួយម៉ែត្រ៖ សម្ពាធសកម្ម Rankine ការផ្ដួលជុំវិញចុង ការរអិល និងសម្ពាធមូលដ្ឋាន។ សម្ពាធអកម្មនៅមុខចុងមិនត្រូវបានរាប់បញ្ចូលឡើយ។ ជំនួយការរបស់អ្នកអាចដំណើរការការពិនិត្យដូចគ្នាដោយប្រើ `pile_group_check` និង `retaining_wall_check`។

ក្រុមបង្គោល ជញ្ជាំង និងការបិទបើកម៉ាត់ត្រូវបានរក្សាទុកក្នុងឯកសារម៉ូដែល ហើយការពិនិត្យដែលបរាជ័យស្ថិតនៅក្នុង {{ui:Problems}}។
