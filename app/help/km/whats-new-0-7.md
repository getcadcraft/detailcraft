នេះជាអ្វីដែលបានផ្លាស់ប្ដូរក្នុង DetailCraft 0.7 បើធៀបនឹង 0.6។ ការវិភាគក្របខ័ណ្ឌនៃ 0.6 ឥឡូវមានជំហានទីពីរ គឺការពិនិត្យសមាជិកដែកតាមកូដរចនា។

## ថ្មី

### ការកាត់មុំ (0.7.1)

- **ធ្នឹម I-shape ពីរជួបគ្នាតាមមុំ** ដូចជាជង្គង់ក្របខ័ណ្ឌ ឬកំពូលដំបូល៖ ជ្រើសរើសទាំងពីរ ហើយជ្រើសរើស {{ui:Mitre}} ក្នុងម៉ឺនុយ {{ui:More}} នៃក្រុម {{ui:Connect}} ចុងទាំងពីរនឹងត្រូវកាត់លើប្លង់ដែលចែកមុំជាពីរ។ បន្ទះចុងដែលភ្ជាប់ដោយប៊ូឡុង និងជញ្ជាំងពង្រឹងគឺជាប្រអប់ធីកក្នុងប្រអប់ Mitre ថ្មី ដែលវាលមានលេខត្រូវនឹងគំនូរនៃការភ្ជាប់របស់អ្នក ដែលបង្កើតឡើងខណៈអ្នកកំពុងវាយ។ ការកាត់មុំធ្វើតាមធ្នឹមនៅពេលវាផ្លាស់ទី ហើយឆ្លងទៅគំនូរ ឯកសារ NC (ការកាត់ទ្រេត) IFC និងបញ្ជីវត្ថុធាតុ។ សូមមើល [ការកាត់មុំ](topic:mitres)។

### ការពិនិត្យការរចនា

- **ការផ្សំបន្ទុកពីកូដ។** នៅក្នុងផ្ទាំង {{ui:Analysis}} ប្លុកថ្មី {{ui:Design checks}} មាន {{ui:Design code}}៖ ASCE 7 + AISC 360 (US, LRFD) ឬ EN 1990 + EN 1993-1-1 (Eurocode)។ ផ្ដល់ប្រភេទឱ្យករណីបន្ទុកនីមួយៗ (ថេរ ចល័ត ដំបូលចល័តថ្មី ព្រិល ខ្យល់ រញ្ជួយដី) ចុច {{ui:Make combinations}} ហើយ DetailCraft នឹងសរសេរការផ្សំកម្លាំង និងការផ្សំភាពប្រើប្រាស់របស់កូដ ចំពោះករណីរបស់អ្នក។ ការផ្សំដែលអ្នកបានវាយដោយខ្លួនឯងនៅតែមាន។ សូមមើល [បង្កើតការផ្សំ](topic:design-checks#make-the-combinations)។
- **តម្លៃ National Annex។** សម្រាប់ Eurocode កត្តាចាប់ផ្ដើមពីតម្លៃដែលបានណែនាំ។ {{ui:National Annex values}} ឱ្យអ្នកវាយតម្លៃរបស់ប្រទេសអ្នក ហើយតម្លៃទាំងនោះត្រូវបានរក្សាទុកក្នុងម៉ូដែល។ សូមមើល [បង្កើតការផ្សំ](topic:design-checks#make-the-combinations)។
- **ពិនិត្យសមាជិក។** {{ui:Check members}} ពិនិត្យធ្នឹម I-shape ដែកគ្រប់ទាំងអស់ចំពោះការទាញ ការបង្ហាប់ជាមួយការខូចរាង ការពត់ជាមួយការខូចរាងចំហៀង-បិទ កម្លាំងកាត់ ការពត់ជាមួយកម្លាំងអ័ក្ស និងការពត់ធៀបនឹង {{ui:Deflection limit}} ដែលអ្នកកំណត់។ តារាងបង្ហាញអត្រាប្រើប្រាស់របស់សមាជិកនីមួយៗ ការពិនិត្យដែលគ្រប់គ្រង និងការផ្សំរបស់វា ហើយសមាជិកត្រូវបានលាបពណ៌លើម៉ូដែល (ខៀវ បៃតង ត្នោតក្រហម ក្រហម ប្រផេះសម្រាប់ដែលមិនបានពិនិត្យ)។ សូមមើល [ពិនិត្យសមាជិក](topic:design-checks#check-the-members)។
- **នៅក្នុងរបាយការណ៍។** {{ui:Report (PDF)}} និង {{ui:Report (CSV)}} បន្ថែមកំណែកូដ ការកំណត់ អត្រាប្រើប្រាស់តាមសមាជិក និងគ្រប់ខណ្ឌទាំងអស់ដែលបានប្រើ។ សូមមើល [របាយការណ៍](topic:design-checks#the-report)។
- **ជំនួយមួយរហូតដល់វិស្វករបានពិនិត្យ។** វាប្រើតែកម្លាំងលំដាប់ទីមួយប៉ុណ្ណោះ។ បំពង់ មុំ ឆានែល ទម្រង់ចំហៀងដែលផ្សារ និងផ្នែកស្តើង ឬថ្នាក់ 4 អានថា «មិនបានពិនិត្យ» ជាមួយហេតុផល។ ការបង្វិល ការភ្ជាប់ និងបេតុងមិនត្រូវបានពិនិត្យទេ។ សូមមើល [អ្វីដែលមិនបានពិនិត្យ](topic:design-checks#what-is-not-checked)។

### USD

- **នាំចេញម៉ូដែលដែកជា USD។** {{ui:USD}} នៅក្នុងក្រុម {{ui:Output}} សរសេរម៉ូដែលជា {{ui:Text (.usda)}}, {{ui:Binary (.usdc)}} ឬ {{ui:Package (.usdz)}} សម្រាប់កម្មវិធីបង្ហាញ និងការផ្គុំឆាក៖ ការផ្គុំនីមួយៗមានផ្នែករបស់វា ផ្នែកនីមួយៗជាផ្នែកកាត់ដែលដាក់តាមប្រវែង ជាមួយការកាត់ឆ្អឹងជ្រុង ហើយនីមួយៗមានសញ្ញាផ្នែក ប្រវែង ថ្នាក់ និងទម្ងន់។ រន្ធប៊ូឡុង ប៊ូឡុង ការផ្សារ និងគ្រឹះមិននៅក្នុងមេ USD ទេ។ សូមមើល [នាំចេញម៉ូដែលដែកជា USD](topic:usd)។

### ជំនួយ AI

- **USD សម្រាប់ជំនួយ។** ជំនួយផ្ទាល់ខ្លួនរបស់អ្នក ដែលភ្ជាប់តាម CadCraft Studio អាចនាំចេញម៉ូដែលជា USD (`usd_export`) ហើយនៅក្នុង Studio សរសេរវាទៅថតដែលអ្នកអនុញ្ញាត។ សូមមើល [អ្វីដែលជំនួយអាចស្នើសុំ](topic:use-with-ai-assistants#what-an-assistant-can-ask-for)។

## បានផ្លាស់ប្ដូរ

- **របារ Steel** មានប៊ូតុងធំ និងតូច៖ ប៊ូតុង {{ui:Beam}} ធំនៅក្បែរ {{ui:Column}}, {{ui:Brace}} និង {{ui:Plate}} តូចនៅលើ {{ui:Modeling}} › {{ui:Members}}; ប៊ូតុង {{ui:End plates}} ធំនៅលើ {{ui:Detailing}} › {{ui:Connections}} ជាមួយ {{ui:Splice}} និង {{ui:Gusset}} ចែករំលែកប៊ូតុងបំបែកតែមួយ; ប៊ូតុង {{ui:Paste}} ធំនៅលើ {{ui:Edit}} › {{ui:Clipboard}} ជាមួយ {{ui:Paste in place}} និង {{ui:Duplicate}} នៅពីក្រោយព្រួញរបស់ក្រុម។ {{ui:Fabrication}} › {{ui:Exchange}} មាន {{ui:Export USD}} នៅក្បែរ IFC។ រចនាបថម៉ឺនុយផ្សេងទៀតនៅដដែល។ សូមមើល [ម៉ឺនុយ គ្រឿងចុច និងកណ្ដុរ](topic:menus-keys-and-mouse#the-three-styles)។
- [វិភាគក្របខ័ណ្ឌ](topic:frame-analysis) ឥឡូវចង្អុលទៅការពិនិត្យការរចនា ហើយការផ្សំរបស់វាអាចត្រូវបានបង្កើតតាមកូដ។

## ក្នុងកម្មវិធីទាំងអស់

- **ឯកតា។** {{ui:Settings}} › {{ui:General}} › {{ui:Units}} មានបញ្ជីមួយសម្រាប់ប្រវែង ផ្ទៃក្រឡា មាឌ មុំ និងជម្រាល ជាមួយទំហំពិតប្រាកដតាមនិយមន័យរបស់ឯកតានីមួយៗ (ហ្វីតសរវេយ US ជាឯកតាដាច់ដោយខ្លួន)។ បិទឯកតាដែលអ្នកមិនដែលប្រើ ហើយវានឹងចេញពីបញ្ជីឯកតា; ជ្រើសមួយជាចំណូលចិត្ត ហើយការងារថ្មីចាប់ផ្ដើមជាមួយវា។ ឯកសារតែងតែរក្សាឯកតារបស់ខ្លួន។ ប្រអប់របស់ DetailCraft នៅមិនទាន់ធ្វើតាមបញ្ជីនេះនៅឡើយ ប្រវែងរបស់វានៅតែវាយជាមីលីម៉ែត្រ។ សូមមើល [ឯកតា តម្លៃដែលបានវាយ និងបញ្ជីឯកតា](common:unit-lists)។
